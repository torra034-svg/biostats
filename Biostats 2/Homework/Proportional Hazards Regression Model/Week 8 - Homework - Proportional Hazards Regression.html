f6 <- f6 %>% 
  mutate(Gender = as.factor(if_else(GENDER == 1, "Male", "Female", missing = NULL)),
         Race = as.factor(case_when(RACE == 1 ~ "Latino",
                          RACE == 2 ~ "African American",
                          RACE == 3 ~ "White",
                          RACE == 4 ~ "Other")),
         Above200 = as.factor(if_else(above200 == 1, "Above 200 cell count", "Below 200 cell count")),
         Randgrp = as.factor(case_when(RANDGRP == 1 ~ "PI + NRTI",
                                       RANDGRP == 2 ~ "NNRTI + NRTI",
                                       RANDGRP == 3 ~ "PI + NNRTI + NRTI")),
         twoclass = as.factor(twoclass),
         TwoClass = as.factor(
           fct_relevel(case_when(twoclass == 1 ~ "Two Class",
                                        twoclass == 2 ~ "Three Class"),"Two Class")))
