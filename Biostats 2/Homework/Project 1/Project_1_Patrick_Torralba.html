Data cleaning

surgery_timing <- surgery_timing %>% 
  mutate(Gender = as.factor(if_else(gender == 1, "Male", "Female")),
         Race = as.factor(case_when(
                          race == 1 ~ "Caucasian",
                          race == 2 ~ "African American",
                          race == 3 ~ "Other")),
         Asa_status = as.factor(case_when(
                                asa_status == 1 ~ "ASA 1-2",
                                asa_status == 2 ~ "ASA 3",
                                asa_status == 3 ~ "ASA 4-6")),
         Complication = factor(if_else(complication == 0, "No Complication", "Complication"),
                               levels = c("No Complication", "Complication")),
         Dow = as.factor(case_when(dow == 1 ~ "Monday",
                         dow == 2 ~ "Tuesday",
                         dow == 3 ~ "Wednesday",
                         dow == 4 ~ "Thursday",
                         dow == 5 ~ "Friday" )),
         Dow = fct_relevel(Dow, "Monday", "Tuesday", "Wednesday", "Thursday", "Friday"))
# this dataset does not have any missing values. This is needed for the likelihood ratio test 

surgery_analysis <- surgery_timing %>% 
  dplyr::select(complication, hour, Dow, age, Gender, bmi, Asa_status) %>% 
  na.omit()

Results

We are doing a logistic regression because the outcome is binary (complication vs no complication).

Hour vs Complication model

Full Model

hour_comp_model_full <- glm(complication ~ hour + age + Gender + bmi + Asa_status, data = surgery_analysis, family = binomial(link = "logit"))

Reduced Model

hour_comp_model_reduced <- glm(complication ~ hour, data = surgery_analysis, family = binomial(link = "logit"))

Linearity Assumption

hour_comp_data <- subset(surgery_analysis, select = c(age, bmi, hour))

hour_comp_predictors <- colnames(hour_comp_data)

hour_comp_data$logit <- predict(hour_comp_model_full)
pivot_longer(hour_comp_data, names_to = "predictors", values_to = "predictor.value", -logit) %>% 
  ggplot(aes(x = predictor.value, y = logit)) +
  geom_point(size = 0.5, alpha = 0.5) +
  geom_smooth(method = "loess", color = "blue", se = FALSE) +
  geom_smooth(method = "lm", color = "red", se = FALSE) +
  facet_wrap(facets = ~ predictors, scales = "free_x") +
  theme_bw()
