surgery_timing <- surgery_timing %>% 
  mutate(Gender = if_else(gender == 1, "Male", "Female"),
         Race = case_when(race == 1 ~ "Caucasian",
                          race == 2 ~ "African American",
                          race == 3 ~ "Other"),
         Asa_status = case_when(asa_status == 1 ~ "ASA 1-2",
                                asa_status == 2 ~ "ASA 3",
                                asa_status == 3 ~ "ASA 4-6"),
         Mort_30 = if_else(mort30 == 0, "No Mortality", "Mortality"))
