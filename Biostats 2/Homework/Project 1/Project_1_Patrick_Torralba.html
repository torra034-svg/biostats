Data cleaning

surgery_timing <- surgery_timing %>% 
  mutate(Gender = as.factor(if_else(gender == 1, "Male", "Female")),
         Race = as.factor(case_when(race == 1 ~ "Caucasian",
                          race == 2 ~ "African American",
                          race == 3 ~ "Other")),
         Asa_status = as.factor(case_when(asa_status == 1 ~ "ASA 1-2",
                                asa_status == 2 ~ "ASA 3",
                                asa_status == 3 ~ "ASA 4-6")),
         Complication = as.factor(if_else(complication == 0, "No Complication", "Complication")),
         Dow = as.factor(case_when(dow == 1 ~ "Monday",
                         dow == 2 ~ "Tuesday",
                         dow == 3 ~ "Wednesday",
                         dow == 4 ~ "Thursday",
                         dow == 5 ~ "Friday" )),
         Dow = fct_relevel(Dow, "Monday", "Tuesday", "Wednesday", "Thursday", "Friday"))

Methods

We are doing a logistic regression because the outcome is binary (complication vs no complication).

Hour vs Complication model

hour_comp_model <- glm(Complication ~ hour + age + Gender + bmi + Asa_status, data = surgery_timing, family = binomial(link = "logit"))
tbl_regression(hour_comp_model, exponentiate = TRUE) %>% 
  bold_labels()
Dow_comp_model <- glm(Complication ~ Dow + age + Gender + bmi + Asa_status, data = surgery_timing, family = binomial(link = "logit"))

Dow vs Complication model
